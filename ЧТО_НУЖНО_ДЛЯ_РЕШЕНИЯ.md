```python
from core.models import User, Photo, TrustFactorHistory

# Найдите волонтера
volunteer = User.objects.filter(role='volunteer').first()

if volunteer:
    print(f"\n=== ВОЛОНТЕР ===")
    print(f"ID: {volunteer.id}")
    print(f"Username: {volunteer.username}")
    print(f"Trust Factor: {volunteer.trust_factor}")
    print(f"Average Rating: {volunteer.average_rating}")
    
    # Последние фото с рейтингом
    photos = Photo.objects.filter(
        volunteer=volunteer, 
        rating__isnull=False
    ).order_by('-uploaded_at')[:5]
    
    print(f"\n=== ПОСЛЕДНИЕ 5 ФОТО ===")
    if photos.exists():
        for p in photos:
            print(f"Photo {p.id}: rating={p.rating}, status={p.status}, uploaded_at={p.uploaded_at}")
    else:
        print("Нет фото с рейтингом")
    
    # История TF
    history = TrustFactorHistory.objects.filter(
        user=volunteer
    ).order_by('-created_at')[:10]
    
    print(f"\n=== ПОСЛЕДНИЕ 10 ИЗМЕНЕНИЙ TF ===")
    if history.exists():
        for h in history:
            print(f"{h.created_at}: {h.old_value} -> {h.new_value} ({h.change_amount:+d}), {h.reason}")
    else:
        print("История изменений пуста")
```
